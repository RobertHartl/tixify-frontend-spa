import { inject } from '@angular/core';
import { CanActivateFn, Router } from '@angular/router';
import { AuthService } from '../services/auth';

export const authGuard: CanActivateFn = () => {
  const auth = inject(AuthService);
  const router = inject(Router);

  if (auth.isLoggedIn()) {
    const role = (auth.currentUser()?.role || '').toLowerCase();
    if (role === 'admin' || role === 'organizer') {
      return true;
    }
  }

  router.navigate(['/login']);
  return false;
};

import { Component, inject } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { HttpErrorResponse } from '@angular/common/http';
import { ApiService } from '../../services/api';

@Component({
  selector: 'app-checkin',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './checkin.html',
  styleUrl: './checkin.css'
})
export class Checkin {
  private api = inject(ApiService);

  ticketCode = '';
  resultStatus: 'idle' | 'valid' | 'used' | 'invalid' = 'idle';
  statusMessage = '';
  isLoading = false;

  onCheckIn(): void {
    const code = this.ticketCode.trim();
    if (!code) return;

    this.isLoading = true;
    this.resultStatus = 'idle';

    this.api.checkInTicket(code).subscribe({
      next: (res: any) => {
        this.isLoading = false;
        this.resultStatus = 'valid';
        this.statusMessage = res.message || 'Ticket erfolgreich entwertet!';
        this.ticketCode = '';
      },
      error: (err: HttpErrorResponse) => {
        this.isLoading = false;
        if (err.status === 409) {
          this.resultStatus = 'used';
          this.statusMessage = 'Ticket wurde bereits entwertet!';
        } else {
          this.resultStatus = 'invalid';
          this.statusMessage = err.error?.error || 'UngÃ¼ltiger Ticket-Code.';
        }
      }
    });
  }
}

import { Component, inject, OnInit, signal, computed } from '@angular/core';
import { CommonModule } from '@angular/common';
import { RouterLink } from '@angular/router';
import { ApiService } from '../../services/api';

@Component({
  selector: 'app-dashboard',
  standalone: true,
  imports: [CommonModule, RouterLink],
  templateUrl: './dashboard.html',
  styleUrl: './dashboard.css'
})
export class Dashboard implements OnInit {
  private api = inject(ApiService);

  isLoading = signal(true);
  events = signal<any[]>([]);

  totalEvents = computed(() => this.events().length);

  totalStock = computed(() => 
    this.events().reduce((sum, e) => sum + (Number(e.event_stock ?? e.stock) || 0), 0)
  );

  totalInventoryValue = computed(() => 
    this.events().reduce((sum, e) => {
      const stock = Number(e.event_stock ?? e.stock) || 0;
      const price = parseFloat(e.price) || 0;
      return sum + (stock * price);
    }, 0)
  );

  avgTicketPrice = computed(() => {
    if (this.totalEvents() === 0) return 0;
    const sumPrice = this.events().reduce((sum, e) => sum + (parseFloat(e.price) || 0), 0);
    return sumPrice / this.totalEvents();
  });

  lowStockEvents = computed(() => 
    this.events().filter(e => (Number(e.event_stock ?? e.stock) || 0) < 50)
  );

  categoryDistribution = computed(() => {
    const counts: Record<string, number> = {};
    for (const e of this.events()) {
      const cat = e.category || 'Sonstige';
      counts[cat] = (counts[cat] || 0) + 1;
    }
    return Object.entries(counts).map(([name, count]) => ({
      name,
      count,
      percent: Math.round((count / (this.totalEvents() || 1)) * 100)
    }));
  });

  ngOnInit(): void {
    this.api.getEvents().subscribe({
      next: (res: any) => {
        const list = Array.isArray(res) ? res : (res.events || res.data || []);
        this.events.set(list);
        this.isLoading.set(false);
      },
      error: (err) => {
        console.error('Fehler beim Laden des Dashboards:', err);
        this.isLoading.set(false);
      }
    });
  }
}

import { Injectable, signal } from '@angular/core';
import { Router } from '@angular/router';

export interface CurrentUser {
  id?: number;
  name?: string;
  email?: string;
  role?: string;
}

@Injectable({
  providedIn: 'root'
})
export class AuthService {
  currentUser = signal<CurrentUser | null>(this.getStoredUser());

  constructor(private router: Router) {}

  private getStoredUser(): CurrentUser | null {
    const raw = localStorage.getItem('user');
    if (raw) {
      try {
        return JSON.parse(raw);
      } catch {
        return null;
      }
    }

    const token = this.getToken();
    if (token) {
      try {
        const payloadBase64 = token.split('.')[1];
        const decoded = JSON.parse(atob(payloadBase64));
        return {
          id: decoded.user_id || decoded.id,
          name: decoded.name || decoded.username || 'Admin',
          email: decoded.email || '',
          role: decoded.role || 'admin'
        };
      } catch {
        return null;
      }
    }
    return null;
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }

  setSession(token: string, user: CurrentUser): void {
    localStorage.setItem('token', token);
    localStorage.setItem('user', JSON.stringify(user));
    this.currentUser.set(user);
  }

  isLoggedIn(): boolean {
    return !!this.getToken();
  }

  getRole(): string | null {
    return this.currentUser()?.role?.toLowerCase() || null;
  }

  isAdmin(): boolean {
    return this.getRole() === 'admin';
  }

  isAdminOrOrganizer(): boolean {
    const role = this.getRole();
    return role === 'admin' || role === 'organizer';
  }

  logout(): void {
    localStorage.removeItem('token');
    localStorage.removeItem('user');
    this.currentUser.set(null);
    this.router.navigate(['/login']);
  }
}
